        // Insert suggestion into editor with proper indentation
        function insertSuggestion(type) {
            if (!window.editor) {
                addTerminalOutput('Editor not initialized', 'error');
                return;
            }

            // Track suggestion clicks
            currentSession.suggestionClickCount++;

            let suggestionText = '';
            if (type === 'guess') {
                suggestionText = document.getElementById('suggestionGuess').textContent.trim();
            } else if (type === 'desc') {
                suggestionText = document.getElementById('suggestionDesc').textContent.trim();
                suggestionText = suggestionText.replace(' // Based on correct problem', '');
            }

            // Don't insert if it's a loading state or empty
            if (!suggestionText || 
                suggestionText.includes('Start coding...') || 
                suggestionText.includes('Based on correct problem...') ||
                suggestionText.includes('No suggestions available') ||
                suggestionText.includes('spinner')) {
                addTerminalOutput('No valid suggestion to insert', 'warning');
                return;
            }

            // Get current cursor position and model
            const position = window.editor.getPosition();
            const model = window.editor.getModel();
            
            // Get current line content and calculate indentation
            const currentLineContent = model.getLineContent(position.lineNumber);
            const currentIndentation = getCurrentIndentation(currentLineContent);
            
            // Determine if we need to add a new line
            const isAtEndOfLine = position.column > currentLineContent.length;
            const needsNewLine = currentLineContent.trim() !== '' || !isAtEndOfLine;
            
            // Process the suggestion text to preserve and apply proper indentation
            let processedSuggestion = processSuggestionWithIndentation(suggestionText, currentIndentation);
            
            // Prepare the final text to insert
            let textToInsert = needsNewLine ? '\n' + processedSuggestion : processedSuggestion;
            
            // For single line suggestions, if we're not at the start of a line, add proper spacing
            if (!needsNewLine && position.column > 1 && !currentLineContent.endsWith(' ')) {
                textToInsert = ' ' + textToInsert;
            }

            // Insert the suggestion using Monaco's command system for better undo support
            window.editor.executeEdits('copilot-suggestion', [{
                range: {
                    startLineNumber: position.lineNumber,
                    startColumn: position.column,
                    endLineNumber: position.lineNumber,
                    endColumn: position.column
                },
                text: textToInsert
            }]);

            // Calculate new cursor position
            const insertedLines = textToInsert.split('\n');
            const newLineNumber = position.lineNumber + insertedLines.length - 1;
            const lastLine = insertedLines[insertedLines.length - 1];
            const newColumn = insertedLines.length === 1 ? 
                position.column + textToInsert.length : 
                lastLine.length + 1;
            
            // Set cursor position and focus
            window.editor.setPosition({
                lineNumber: newLineNumber,
                column: newColumn
            });
            window.editor.focus();

            // Trigger formatting if it's a complete statement
            if (suggestionText.includes('{') || suggestionText.includes('}') || suggestionText.includes(';')) {
                setTimeout(() => {
                    window.editor.getAction('editor.action.formatDocument')?.run();
                }, 100);
            }

            // Log the insertion
            const sourceType = type === 'guess' ? 'AI Problem Guess' : 'Problem Description';
            addTerminalOutput(`Inserted suggestion from ${sourceType}`, 'success');
            const preview = suggestionText.split('\n')[0].substring(0, 40);
            addTerminalOutput(`Code: ${preview}${suggestionText.length > 40 ? '...' : ''}`, 'info');
        }

        // Helper function to get current line indentation
        function getCurrentIndentation(lineContent) {
            const match = lineContent.match(/^(\s*)/);
            return match ? match[1] : '';
        }

        // Helper function to process suggestion with proper indentation
        function processSuggestionWithIndentation(suggestion, baseIndentation) {
            const lines = suggestion.split('\n');
            if (lines.length === 1) {
                return suggestion.trim();
            }

            const processedLines = [];
            let additionalIndent = '';
            
            for (let i = 0; i < lines.length; i++) {
                const line = lines[i].trim();
                if (line === '') {
                    processedLines.push('');
                    continue;
                }

                if (i === 0) {
                    processedLines.push(baseIndentation + line);
                } else {
                    const prevLine = lines[i - 1].trim();
                    
                    if (prevLine.endsWith('{') || 
                        prevLine.endsWith('(') ||
                        prevLine.includes('if ') ||
                        prevLine.includes('for ') ||
                        prevLine.includes('while ') ||
                        prevLine.includes('function') ||
                        prevLine.includes('=>')) {
                        additionalIndent = '    ';
                    }
                    
                    if (line.startsWith('}') || line.startsWith(')')) {
                        additionalIndent = '';
                    }
                    
                    processedLines.push(baseIndentation + additionalIndent + line);
                }
            }

            return processedLines.join('\n');
        }

        // Initialize Monaco Editor - FIXED: Better error handling
        async function initializeEditor() {
            return new Promise((resolve, reject) => {
                if (window.editor) {
                    try {
                        window.editor.dispose();
                        window.editor = null;
                    } catch (e) {
                        console.log('Previous editor cleanup:', e);
                    }
                }
                
                const editorContainer = document.getElementById('editor');
                if (!editorContainer) {
                    reject(new Error('Editor container not found'));
                    return;
                }
                
                editorContainer.innerHTML = '';
                
                require.config({ paths: { 'vs': 'https://cdnjs.cloudflare.com/ajax/libs/monaco-editor/0.43.0/min/vs' }});
                require(['vs/editor/editor.main'], function () {
                    try {
                        window.editor = monaco.editor.create(editorContainer, {
                            value: '// Start coding here...\n',
                            language: 'javascript',
                            theme: 'vs-dark',
                            automaticLayout: true,
                            minimap: { enabled: true },
                            scrollBeyondLastLine: false,
                            wordWrap: 'on',
                            lineNumbers: 'on',
                            folding: true,
                            selectOnLineNumbers: true,
                            matchBrackets: 'always',
                            fontSize: currentFontSize
                        });

                        // AI analysis triggers on meaningful content changes with debouncing
                        let analysisTimeout;
                        let lastAnalyzedContent = '';
                        
                        window.editor.onDidChangeModelContent(() => {
                            // Clear existing timeout to prevent multiple rapid calls
                            clearTimeout(analysisTimeout);
                            
                            // Set timeout for analysis - only triggers after user stops making changes
                            analysisTimeout = setTimeout(() => {
                                const currentContent = window.editor.getValue().trim();
                                
                                // Only analyze if content actually changed meaningfully
                                if (currentContent !== lastAnalyzedContent && currentContent.length > 0) {
                                    analyzeCurrentCode();
                                    lastAnalyzedContent = currentContent;
                                }
                            }, 2000); // 2-second delay to avoid constant analysis
                        });
                        
                        updateFontSize(currentFontSize);
                        resolve();
                    } catch (error) {
                        console.error('Monaco Editor initialization failed:', error);
                        reject(error);
                    }
                }, function (err) {
                    console.error('Monaco Editor loading failed:', err);
                    reject(new Error('Failed to load Monaco Editor'));
                });
            });
        }

        async function analyzeCurrentCode() {
            if (!currentSession.sessionId) {
                console.warn('No active session for code analysis');
                return;
            }
            
            const code = window.editor.getValue().trim();
            if (!code || code === '// Start coding here...') return;

            try {
                const response = await fetch('/analyze', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({
                        code: code,
                        user_id: currentSession.userId,
                        assignment_key: currentSession.assignmentKey,
                        session_id: currentSession.sessionId
                    })
                });

                if (!response.ok) {
                    throw new Error(`Analysis failed: ${response.statusText}`);
                }

                const data = await response.json();
                const problemDescription = data.analysis || "Problem description not available";
                let nextLineCode = data.next_line || "// No suggestions available";

                // Clean and format suggested code
                nextLineCode = nextLineCode.replace(/^```\w*\n?|```$/g, '').trim();
                const lines = nextLineCode.split('\n');
                if (lines.length > 2) {
                    nextLineCode = lines.slice(0, 2).join('\n');
                }

                analysis.textContent = problemDescription;
                document.getElementById('suggestionGuess').textContent = nextLineCode;

                // FIXED: Store in guess history with consistent field names
                currentSession.guessHistory.push({
                    timestamp: new Date().toISOString(),
                    problemAnalysis: problemDescription, // FIXED: Consistent field name
                    nextLine: nextLineCode
                });

                // Update log counts
                currentSession.guessLogCount++;
                guessLogCount.textContent = currentSession.guessLogCount;

                // For condition 1, also update description-based suggestions
                if (currentSession.condition === 1) {
                    const descBasedSuggestion = nextLineCode; // Remove the comment addition
                    document.getElementById('suggestionDesc').textContent = descBasedSuggestion;
                    
                    // FIXED: Store in description history with consistent field names
                    currentSession.descHistory.push({
                        timestamp: new Date().toISOString(),
                        problemAnalysis: currentSession.problemDescription || problemDescription, // Use stored problem description
                        nextLine: descBasedSuggestion
                    });
                    
                    currentSession.descLogCount++;
                    descLogCount.textContent = currentSession.descLogCount;
                }

            } catch (error) {
                console.error('Analysis error:', error);
                analysis.innerHTML = `
                    <div class="error-message">
                        <h3>Connection Issue</h3>
                        <p><strong>Error:</strong> ${error.message}</p>
                        <p>Local logging continues...</p>
                    </div>
                `;
            }
        }

        // Submit to Database and FORCE LOGOUT
        async function submitToDatabase() {
            if (!currentSession.sessionId) {
                addTerminalOutput('No active session to submit', 'error');
                return;
            }

            const code = window.editor.getValue();
            if (!code.trim()) {
                addTerminalOutput('No code to submit. Please write some code first.', 'warning');
                return;
            }

            // Confirm submission with logout warning
            const confirmSubmit = confirm(
                'IMPORTANT: Submitting your code will automatically log you out of this session.\n\n' +
                'You will need to start a new session with your credentials to continue coding.\n\n' +
                'Are you sure you want to submit and logout?'
            );

            if (!confirmSubmit) {
                addTerminalOutput('Submission cancelled by user', 'warning');
                return;
            }

            submitButton.disabled = true;
            submitButton.innerHTML = '<div class="terminal-loading"><span class="terminal-spinner"></span>Submitting & Logging out...</div>';

            try {
                const response = await fetch('/submit', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({
                        code: code,
                        user_id: currentSession.userId,
                        assignment_key: currentSession.assignmentKey,
                        session_id: currentSession.sessionId,
                        timestamp: new Date().toISOString(),
                        suggestion_clicked: currentSession.suggestionClickCount,
                        compiler_clicked: currentSession.compilerClickCount
                    })
                });

                if (!response.ok) {
                    throw new Error(`Submission failed: ${response.statusText}`);
                }

                const result = await response.json();
                addTerminalOutput(`✅ ${result.message}`, 'success');
                addTerminalOutput(`🔥 Database Status: ${result.firebase_status}`, 'info');
                addTerminalOutput(`📊 Submission ID: ${result.submission_id}`, 'info');
                addTerminalOutput(`📈 Problem Guess Logs: ${currentSession.guessLogCount}`, 'info');
                addTerminalOutput(`🖱️ Suggestions Clicked: ${currentSession.suggestionClickCount}`, 'info');
                addTerminalOutput(`⚡ Compiler Runs: ${currentSession.compilerClickCount}`, 'info');
                if (currentSession.condition === 1) {
                    addTerminalOutput(`📋 Description-based Logs: ${currentSession.descLogCount}`, 'info');
                }
                
                // Check if Firebase save was successful before logging out
                if (result.firebase_status === 'success') {
                    addTerminalOutput(`🔓 Database save successful - logging out now...`, 'success');
                    setTimeout(() => {
                        showSuccessMessage(
                            'Code submitted successfully to Database!\nSession ended for security.', 
                            3
                        );
                    }, 500);
                } else {
                    addTerminalOutput(`⚠️ Database save failed but data preserved locally - logging out...`, 'warning');
                    setTimeout(() => {
                        showSuccessMessage(
                            'Submission completed with local backup.\nSession ended for security.', 
                            3
                        );
                    }, 500);
                }

            } catch (error) {
                console.error('Submission error:', error);
                addTerminalOutput(`❌ Submission failed: ${error.message}`, 'error');
                addTerminalOutput(`💾 Data preserved locally in session logs`, 'warning');
                submitButton.disabled = false;
                submitButton.innerHTML = '🔥 Submit & Logout';
            }
        }

        // End session manually
        function endSession() {
            if (confirm('Are you sure you want to end this session? Any unsaved code will be lost.')) {
                addTerminalOutput('🔓 Session ended by user', 'warning');
                showSuccessMessage('Session ended. Redirecting to login...', 2);
            }
        }

        // Add output to terminal
        function addTerminalOutput(text, type = 'normal') {
            const line = document.createElement('div');
            line.className = 'output-line';
            
            switch (type) {
                case 'success':
                    line.innerHTML = `<span class="output-success">${text}</span>`;
                    break;
                case 'error':
                    line.innerHTML = `<span class="output-error">${text}</span>`;
                    break;
                case 'info':
                    line.innerHTML = `<span class="output-info">${text}</span>`;
                    break;
                case 'warning':
                    line.innerHTML = `<span class="output-warning">${text}</span>`;
                    break;
                case 'prompt':
                    line.innerHTML = `<span class="output-prompt">>>> </span>${text}`;
                    break;
                case 'execution-time':
                    line.innerHTML = `<span class="execution-time">${text}</span>`;
                    break;
                default:
                    line.innerHTML = `<pre>${text}</pre>`;
            }
            
            const terminalOutput = document.getElementById('terminalOutput');
            terminalOutput.appendChild(line);
            terminalOutput.scrollTop = terminalOutput.scrollHeight;
        }

        // Clear terminal
        function clearTerminal() {
            const terminalOutput = document.getElementById('terminalOutput');

                </div>

                    <span class="output-prompt">>>> </span><span class="output-info">Continue coding!</span>
                </div>
            `;
        }

        // Keyboard shortcuts for font size
        document.addEventListener('keydown', (e) => {
            if ((e.ctrlKey || e.metaKey) && (e.key === '=' || e.key === '+')) {
                e.preventDefault();
                if (currentFontSize < maxFontSize) {
                    increaseFontSize();
                }
            }
            if ((e.ctrlKey || e.metaKey) && e.key === '-') {
                e.preventDefault();
                if (currentFontSize > minFontSize) {
                    decreaseFontSize();
                }
            }
            if ((e.ctrlKey || e.metaKey) && e.key === '0') {
                e.preventDefault();
                updateFontSize(14);
                addTerminalOutput(`Font size reset to ${currentFontSize}px`, 'info');
            }
        });

        // Resize handling
        const resize = {};
        resize.sidebar = false;
        resize.terminal = false;
        
        document.getElementById('resize-sidebar').addEventListener('mousedown', (e) => {
            resize.sidebar = true;
            document.addEventListener('mousemove', handleSidebarResize);
            document.addEventListener('mouseup', handleMouseUp);
        });

        document.getElementById('resize-terminal').addEventListener('mousedown', (e) => {
            resize.terminal = true;
            document.addEventListener('mousemove', handleTerminalResize);
            document.addEventListener('mouseup', handleMouseUp);
        });

        function handleSidebarResize(e) {
            if (!resize.sidebar) return;
            document.getElementById('sidebar').style.width = `${window.innerWidth - e.clientX}px`;
        }

        function handleTerminalResize(e) {
            if (!resize.terminal) return;
            document.getElementById('terminal').style.height = `${window.innerHeight - e.clientY}px`;
        }

        function handleMouseUp() {
            resize.sidebar = false;
            resize.terminal = false;
            document.removeEventListener('mouseup', handleMouseUp);
            document.removeEventListener('mousemove', handleSidebarResize);
            document.removeEventListener('mousemove', handleTerminalResize);
        }

        // Prevent accidental page close
        window.addEventListener('beforeunload', (e) => {
            if (currentSession.sessionId) {
                e.preventDefault();
                e.returnValue = 'You have an active secure coding session! Submit your code to Database before leaving to save your work.';
                return e.returnValue;
            }
        });

        // Clear form on page load
        window.addEventListener('load', () => {
            if (!authScreen.classList.contains('hidden')) {
                document.getElementById('userId').value = '';
                document.getElementById('assignmentKey').value = '';
            }
        });
